cmake_minimum_required(VERSION 3.16)
project(tetris VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(game_core src/tetris.c)
target_include_directories(game_core PUBLIC include)
add_executable(tetris src/main.c)
target_link_libraries(tetris PRIVATE game_core)
if(MSVC)
  target_compile_options(game_core PRIVATE /W4 /utf-8)
  target_compile_options(tetris PRIVATE /W4 /utf-8)
else()
  target_compile_options(game_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(tetris PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_core tests/test_core.c)
  target_link_libraries(test_core PRIVATE game_core)
  # Tests use assert even in Release configurations.
  if(MSVC)
    target_compile_options(test_core PRIVATE /UNDEBUG)
  else()
    target_compile_options(test_core PRIVATE -UNDEBUG)
  endif()
  add_test(NAME core COMMAND test_core)
endif()
